871:   // Utility: draw pixel art from string array
872:   drawPixelArt(graphics, data, colorMap, pixelSize, offsetX, offsetY) {
873:     for (let row = 0; row < data.length; row++) {
874:       for (let col = 0; col < data[row].length; col++) {
875:         const ch = data[row][col];
876:         if (colorMap[ch] !== undefined && colorMap[ch] !== -1) {
877:           graphics.fillStyle(colorMap[ch], 1);
878:           graphics.fillRect(
879:             offsetX + col * pixelSize,
880:             offsetY + row * pixelSize,
881:             pixelSize,
882:             pixelSize
883:           );
884:         }
885:       }
886:     }
887:   }
